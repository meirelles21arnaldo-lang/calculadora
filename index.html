function calcularCompatibilidade(nome1, nome2) { let n1 = removerAcentos(nome1).toUpperCase().trim(); let n2 = removerAcentos(nome2).toUpperCase().trim(); let nomesOrdenados = [n1, n2].sort(); let nomeJunto = nomesOrdenados.join("").replace(/ /g, ""); let frequencias = {}; let ordemUnica = []; for (let char of nomeJunto) { if (!(char in frequencias)) { frequencias[char] = 0; ordemUnica.push(char); } frequencias[char]++; } let vetor = ordemUnica.map(char => frequencias[char]); while (vetor.length > 2) { let novoVetor = []; let i = 0, j = vetor.length - 1; while (i < j) { novoVetor.push(vetor[i] + vetor[j]); i++; j--; } if (i === j) novoVetor.push(vetor[i]); vetor = novoVetor; } // Calcula o resultado base a partir do vetor let resultado = 0; if (vetor.length === 2) { resultado = parseInt(`${vetor[0]}${vetor[1]}`); } else { resultado = vetor[0]; } // Se der mais de 100, aplica a lógica de redução com base nos dígitos while (resultado > 100) { let stringFinal = resultado.toString(); if (stringFinal.length === 3) { let d1 = parseInt(stringFinal[0]); let d2 = stringFinal[1]; let d3 = parseInt(stringFinal[2]); resultado = parseInt(`${d1 + d3}${d2}`); } else { // Caso tenha 4 dígitos ou mais, reduz juntando o vetor original resultado = parseInt(vetor.join("")); break; } } let msg = ""; if (resultado < 10) msg = "Ugh, pouco compatível"; else if (resultado <= 25) msg = "Pouca chance"; else if (resultado < 50) msg = "Eu tentaria outro"; else if (resultado === 50) msg = "Cara ou coroa? (Do meu jeito)"; else if (resultado <= 75) msg = "Com um pouco de esforço funciona, não desista!"; else if (resultado <= 90) msg = "Pouco esforço, é só querer!"; else msg = "Combinação quase perfeita! Vá em frente!"; return { porcentagem: resultado, mensagem: msg }; }
